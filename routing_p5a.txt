        );
        if (r.ok) { const d = await r.json(); if (d.display_name) shortName = d.display_name.split(',').slice(0, 2).join(', '); }
      } catch {}
      const place = { lat, lng, name: shortName, shortName };

      else { setTo(place); setToText(shortName); }
